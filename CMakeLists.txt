cmake_minimum_required(VERSION 3.28)
project(stream-checklist VERSION 1.0.0 LANGUAGES C CXX)
set(CMAKE_CXX_STANDARD 17)
set(CMAKE_CXX_STANDARD_REQUIRED ON)
option(BUILD_PLUGIN "Build the native OBS plugin" ON)
include(CTest)
if(BUILD_TESTING)
  add_executable(checklist-tests tests/checks.cpp)
  target_include_directories(checklist-tests PRIVATE src)
  add_test(NAME checklist-status COMMAND checklist-tests)
endif()
if(BUILD_PLUGIN)
  find_package(libobs REQUIRED CONFIG)
  find_package(obs-frontend-api REQUIRED CONFIG)
  find_package(Qt6 REQUIRED COMPONENTS Widgets)
  add_library(stream-checklist MODULE src/plugin.cpp)
  target_link_libraries(stream-checklist PRIVATE OBS::libobs OBS::obs-frontend-api Qt6::Widgets)
  set_target_properties(stream-checklist PROPERTIES PREFIX "")
  if(MSVC)
    target_compile_options(stream-checklist PRIVATE /W4 /utf-8)
  else()
    target_compile_options(stream-checklist PRIVATE -Wall -Wextra)
  endif()
  install(TARGETS stream-checklist RUNTIME DESTINATION stream-checklist/bin/64bit LIBRARY DESTINATION stream-checklist/bin/64bit)
  install(FILES README.md DESTINATION stream-checklist/data)
endif()
